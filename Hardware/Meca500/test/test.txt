;此版本代码仅用于机械臂调试动作

; =============================================================================
; 第 1 节: 标定点位 (所有值定义在这里, 改值即生效)
; =============================================================================

; ---- 1.1 Home 关节角 (deg) ----
CreateVariable(home.joints,  [0.0, -40.0, 0.0, 0.0, 40.0, 90.0], 0, 1)

; ---- 1.2 预抓取位 (TCP 在抓取位后方, 远离工件) ----
CreateVariable(pre.pick,     [297.0, -134.707, 308.0, 0.0, 90.0, 90.0], 0, 1)

; ---- 1.3 抓取位 (TCP 在板面) ----
CreateVariable(pick.pose,    [327.0, -134.707, 308.0, 0.0, 90.0, 90.0], 0, 1)

; ---- 1.4 预放置位 (TCP 在放置位后方, 远离设备) ----
CreateVariable(pre.place,    [197.0, 106.847, 360.997, 0.0, 90.0, 90.0], 0, 1)

; ---- 1.5 放置位 ----
CreateVariable(place.pose,   [227.0, 106.847, 360.997, 0.0, 90.0, 90.0], 0, 1)

; ---- 1.6 回程路点 (PrePlace → Home 中途) ----
CreateVariable(way.point,     [195.0, 50.000, 350.0, 0.0, 90.0, 90.0], 0, 1)

; ---- 1.7 夹爪等待时间 (s) ----
CreateVariable(time.grip,    0.3, 0, 1)


; =============================================================================
; 第 2 节: 初始化
; =============================================================================
ActivateRobot()
SetJointVel(10)
SetCartLinVel(5)
SetCartAcc(200)
SetGripperForce(30)
SetGripperVel(50)

Home()
SetConf(1, 1, 1)
GripperOpen()


; =============================================================================
; 第 3 节: Pick-and-Place 主循环
; =============================================================================
;  *vars.<name>  展开数组
;  vars.<name>   单值变量
;
;  退避位 = 预抓取/预放置位 (Pre → Pick → Pre → Place → Pre → Home)

; ---- Step 1: Home → PrePick (PTP) ----
SetBlending(100)
MoveJoints(*vars.home.joints)
MovePose(*vars.pre.pick)

; ---- Step 2: PrePick → Pick (直线接近) ----
SetBlending(0)
MoveLin(*vars.pick.pose)

; ---- Step 3: 抓取 ----
GripperClose()
Delay(vars.time.grip)

; ---- Step 4: Pick → PrePick (直线退避) ----
SetBlending(0)
MoveLin(*vars.pre.pick)

; ---- Step 5: PrePick → PrePlace (PTP, α/β 恒定起止一致) ----
SetBlending(100)
MovePose(*vars.pre.place)

; ---- Step 6: PrePlace → Place (直线接近) ----
SetBlending(0)
MoveLin(*vars.place.pose)

; ---- Step 7: 放置 ----
GripperOpen()
Delay(vars.time.grip)

; ---- Step 8: Place → PrePlace (直线退避) ----
SetBlending(0)
MoveLin(*vars.pre.place)

; ---- Step 9: PrePlace → Waypoint → Home ----
SetBlending(100)
MovePose(*vars.way.point)
MoveJoints(*vars.home.joints)


; =============================================================================
; 附录: 标定变量速查表
; =============================================================================
;
; ┌──────────────────────┬───────────────────────────────────────────────────┐
; │ 变量                 │ 值                                               │
; ├──────────────────────┼───────────────────────────────────────────────────┤
; │ vars.home.joints     │ [0, -40, 0, 0, 40, 90] (关节角 deg)           │
; │ vars.pre.pick        │ [297, -134.707, 308, 0, 90, 90]                 │
; │ vars.pick.pose       │ [327, -134.707, 308, 0, 90, 90]                 │
; │ vars.pre.place       │ [197, 106.847, 360.997, 0, 90, 90]              │
; │ vars.place.pose      │ [227, 106.847, 360.997, 0, 90, 90]              │
; │ vars.way.point       │ [195, 50, 350, 0, 90, 90] (回程路点)            │
; │ vars.time.grip       │ 0.3 (夹爪等待 s)                                 │
; └──────────────────────┴───────────────────────────────────────────────────┘
;
; 重新标定: 修改第 1 节数值或 MecaPortal Variables 面板直接改
; 安全: 首次运行降速 SetJointVel(10) / SetCartLinVel(5)
